Returns "" for non-work pages. Context: the page. Returns an HTML string. */}} {{ $out := "" }} {{ if eq .Params.doctype "Work" }} {{ $stem := .File.ContentBaseName }} {{ $key := "" }} {{ range $k, $_ := site.Data.editions }} {{ if and (or (eq $stem $k) (hasPrefix $stem (printf "%s-" $k))) (gt (len $k) (len $key)) }} {{ $key = $k }} {{ end }} {{ end }} {{ $credit := "" }} {{ with $key }} {{ $e := index site.Data.editions . }} {{ if $e.credit }} {{ $credit = $e.credit }} {{ else }} {{ $credit = printf "Edited by %s (%v%s)." $e.editors $e.year (cond (isset $e "extra") (printf ", %s" $e.extra) "") }} {{ end }} {{ $credit = printf `%s ` $credit }} {{ end }} {{ $note := `Arranged as in the published edition, but with Wittgenstein’s original text, which differs from the printed book in variants, spelling and punctuation.` }} {{ $out = printf `

%s%s

` $credit $note }} {{ end }} {{ return $out }}
