{{/* Editors line + note for work pages: who edited the published book whose arrangement the page follows (site/data/editions.yaml), on a line of its own a revised edition or a later title if there is one, and why the text differs from that book. The wording is in site/data/edition_note.yaml, shared with the ebooks and pdfs (tools/editions.rs). Returns "" for non-work pages. Context: the page. Returns an HTML string. */}} {{ $out := "" }} {{ $wording := site.Data.edition_note }} {{ if eq .Params.doctype "Work" }} {{ $stem := .File.ContentBaseName }} {{ $key := "" }} {{ range $k, $_ := site.Data.editions }} {{ if and (or (eq $stem $k) (hasPrefix $stem (printf "%s-" $k))) (gt (len $k) (len $key)) }} {{ $key = $k }} {{ end }} {{ end }} {{ $credit := "" }} {{ with $key }} {{ $e := index site.Data.editions . }} {{ if $e.credit }} {{ $credit = $e.credit }} {{ else }} {{ $extra := cond (isset $e "extra") (printf ", %s" $e.extra) "" }} {{ $credit = replace (replace (replace $wording.credit "{editors}" $e.editors) "{year}" (string $e.year)) "{extra}" $extra }} {{ end }} {{ $credit = printf `%s ` $credit }} {{ with $e.revision }} {{ $credit = printf `%s%s ` $credit . }} {{ end }} {{ end }} {{ $note := replace $wording.note "{about}" "/about/#editorial-criteria" }} {{ $out = printf `

%s%s

` $credit $note }} {{ end }} {{ return $out }}
